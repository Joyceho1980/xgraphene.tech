class IngestionPipeline:
    def fetch_url(self, url):
        html = requests.get(url).text
        soup = BeautifulSoup(html, "html.parser")
        return soup.get_text()

    def clean_text(self, text):
        return " ".join(text.split())

from engine.embedder import Embedder
import itertools

class AutoGraphBuilder:
    def __init__(self):
        self.embedder = Embedder()

    def extract_nodes(self, text_chunks):
        nodes = []

        for chunk in text_chunks:
            nodes.append({
                "id": self._make_id(chunk),
                "definition": chunk,
                "ai_citable": chunk
            })

        return nodes

    def extract_edges(self, nodes):
        edges = []

        embeddings = self.embedder.encode([n["definition"] for n in nodes])

        for i, j in itertools.combinations(range(len(nodes)), 2):
            sim = float(
                (embeddings[i] @ embeddings[j]) /
                (abs(embeddings[i]) * abs(embeddings[j]) + 1e-9)
            )

            if sim > 0.55:
                edges.append({
                    "from": nodes[i]["id"],
                    "to": nodes[j]["id"],
                    "relation": "correlates_with",
                    "weight": sim
                })

        return edges

    def _make_id(self, text):
        return str(abs(hash(text)) % 10**8)

class HybridRetriever:
    def __init__(self, rag_engine, graph):
        self.rag = rag_engine
        self.graph = graph

    def retrieve(self, query):
        # 1. semantic retrieval
        base_nodes = self.rag.retrieve(query)

        # 2. graph expansion
        expanded = set(base_nodes)

        for node in base_nodes:
            expanded.update(
                self.graph.expand(node, depth=1)
            )

        return list(expanded)

class RAGEngineV3:
    def __init__(self, retriever, nodes):
        self.retriever = retriever
        self.nodes = nodes

    def generate_answer(self, query):
        nodes = self.retriever.retrieve(query)

        context = []
        for n in nodes:
            node = self._get_node(n)
            if node:
                context.append(node["ai_citable"])

        return {
            "query": query,
            "context": context,
            "answer": self._compose(context)
        }

    def _compose(self, context):
        return " ".join(context)

class SEOExporter:
    def generate_ai_snippet(self, node):
        return {
            "title": node["id"],
            "ai_summary": node["ai_citable"],
            "schema": {
                "@type": "Dataset",
                "name": node["id"],
                "text": node["ai_citable"]
            }
        }

    def generate_sitemap_node(self, nodes):
        return [
            {
                "url": f"/node/{n['id']}",
                "priority": 0.8
            }
            for n in nodes
        ]
