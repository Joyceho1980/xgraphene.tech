class GEOBatchPipeline:

    def run(self, articles, hub):
        for article in articles:

            schema = SchemaGenerator().generate(

                nodes=["Mitochondria", "Cellular Energy"],
                url=article["file"]
            )

            html = open(article["file"]).read()

            html = HTMLInjector().inject_ai_card(html, assertion, context)
            html = HTMLInjector().inject_schema(html, schema)

            save(article["file"], html)
