class QueryParser:
    def parse(self, query):
        return {
            "intent": self.detect_intent(query),
            "entities": self.extract_entities(query),
            "depth": self.detect_depth(query),
            "domain": self.detect_domain(query)
        }

    def detect_intent(self, q):
        if "why" in q.lower():
            return "causal_explanation"
        if "how" in q.lower():
            return "process_explanation"
        return "informational"

    def detect_depth(self, q):
        if "deep" in q.lower():
            return "deep"
        return "standard"

class SearchOrchestrator:
    def route(self, query):
        intent = query["intent"]

        if intent == "causal_explanation":
            return "graph_rag"
        if intent == "fact_lookup":
            return "web_search"
        return "hybrid"

class CitationSimulator:
    def predict_citations(self, query, graph):
        candidates = graph.retrieve(query)

        ranked = sorted(
            candidates,
            key=lambda x: x["authority_score"],
            reverse=True
        )

        return ranked[:5]
