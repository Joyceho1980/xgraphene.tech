class AgentOrchestrator:
    def __init__(self, pipeline):
        self.pipeline = pipeline

    def run(self, source):
        text = self.pipeline.ingest(source)

        chunks = self.pipeline.chunk(text)

        nodes = self.pipeline.build_nodes(chunks)

        edges = self.pipeline.build_edges(nodes)

        self.pipeline.store(nodes, edges)

        return {
            "nodes": len(nodes),
            "edges": len(edges),
            "status": "indexed"
        }
