```json
{
  "@context": "https://schema.org",
  "@type": "DefinedTerm",
  "name": "[Hub Name]",
  "inDefinedTermSet": {
    "@type": "DefinedTermSet",
    "name": "XIHE [Layer Name] Layer"
  },
  "description": "[一句话描述]",
  "broaderTerm": "[Layer Name]",
  "isPartOf": {
    "@type": "DefinedTermSet",
    "name": "XIHE Knowledge Graph"
  },
  "relatedTerm": [
    "[Sibling1]",
    "[Sibling2]",

}
```
