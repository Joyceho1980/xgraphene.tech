class CitationTracker:
    def __init__(self):
        self.citations = {}

    def log(self, source, node):
        if node not in self.citations:
            self.citations[node] = []

        self.citations[node].append(source)

    def get_authority_score(self, node):
        return len(self.citations.get(node, []))

            "citation_ready": True,
