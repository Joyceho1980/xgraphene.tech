class GraphEconomy:
    def edge_value(self, edge):
        return edge["weight"] * 10

    def node_value(self, graph, node):
        incoming = graph.in_degree(node, weight="weight")
        outgoing = graph.out_degree(node, weight="weight")

        return incoming * 1.5 + outgoing * 0.5

class InformationMarket:
    def price_node(self, node):
        return (
            node.citation_score * 0.4 +
            node.search_volume * 0.4 +
            node.conversion_rate * 0.2
        )

    def market_rank(self, nodes):
        return sorted(nodes, key=lambda x: self.price_node(x), reverse=True)
